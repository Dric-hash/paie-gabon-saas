🔐 Authentification

# Token fixe
curl -H "X-API-Key: VOTRE_TOKEN" \
  {{ request.host_url }}api/v1/me

# OAuth2
curl -X POST {{ request.host_url }}api/v1/oauth/token \
  -H "Content-Type: application/json" \
  -d '{"grant_type":"client_credentials",
       "client_id":"pg_xxx",
       "client_secret":"yyy"}'
